function compression_gui
    % ------- 전역 변수 -------
    global arduino cas running startTime targetDuration ...
           distanceData forceData csvFile t plotTimer;

    arduino_port = "COM3";
    cas_port = "COM4";
    baudrate = 9600;
    running = false;

    % ------- 시리얼 포트 연결 -------
    try
        arduino = serialport(arduino_port, 9600);
        pause(2);
        disp("✅ Arduino 연결 완료");
    catch
        disp("❌ Arduino 연결 실패");
        arduino = [];
    end

    try
        cas = serialport(cas_port, baudrate);
        configureTerminator(cas, "LF");   % 센서가 \n 기준일 경우
        flush(cas);
        pause(2);
        disp("✅ CAS 센서 연결 완료");
    catch
        disp("❌ CAS 센서 연결 실패");
        cas = [];
    end

    % ------- GUI 구성 -------
    fig = uifigure('Name', 'Compression Control', 'Position', [100 100 700 400]);

    % 입력
    lblDist = uilabel(fig, 'Text', '거리 (mm):', 'Position', [20 350 80 22]);
    txtDist = uieditfield(fig, 'numeric', 'Position', [110 350 100 22]);

    lblSpeed = uilabel(fig, 'Text', '속도 (mm/s):', 'Position', [20 310 80 22]);
    txtSpeed = uieditfield(fig, 'numeric', 'Position', [110 310 100 22]);

    lblCorr = uilabel(fig, 'Text', '보정계수:', 'Position', [20 270 80 22]);
    txtCorr = uieditfield(fig, 'numeric', 'Value', 1.0, 'Position', [110 270 100 22]);

    % 시간 표시
    lblTime = uilabel(fig, 'Text', '실제 소요시간: - s', 'Position', [20 230 200 22]);

    % 버튼 (Start, Stop, Back)
    btnStart = uibutton(fig, 'Text', 'Start', 'Position', [20 190 80 30], ...
        'ButtonPushedFcn', @(btn, event) startCompression());
    btnStop = uibutton(fig, 'Text', 'Stop', 'Position', [120 190 80 30], ...
        'ButtonPushedFcn', @(btn, event) stopCompression());
    btnBack = uibutton(fig, 'Text', 'Back', 'Position', [20 150 180 30], ...
        'ButtonPushedFcn', @(btn, event) backMove());

    % 그래프
    ax = uiaxes(fig, 'Position', [250 50 420 320]);
    title(ax, 'Distance vs Force (N)');
    xlabel(ax, 'Distance (mm)');
    ylabel(ax, 'Force (N)');
    grid(ax, 'on');
    plt = animatedline(ax, 'Color', 'b');

    % Plot 타이머 설정
    plotTimer = timer('ExecutionMode', 'fixedRate', 'Period', 0.2, ...
        'TimerFcn', @(~,~) updatePlot());

    % ------- 압축 시작 함수 -------
    function startCompression()
        dist = txtDist.Value;
        speed = txtSpeed.Value;
        corr = txtCorr.Value;

        if isempty(dist) || isempty(speed)
            uialert(fig, '거리와 속도를 입력하세요', '입력 오류');
            return;
        end

        targetDuration = dist / speed;
        startTime = tic;
        running = true;
        distanceData = [];
        forceData = [];

        % CSV 파일 생성
        nowStr = datestr(now, 'yyyymmdd_HHMMSS');
        csvFile = [' ' nowStr '.csv'];
        writematrix(["Time (s)", "Distance (mm)", "Force (N)"], csvFile);

        % 아두이노 명령 전송
        if ~isempty(arduino)
            writeline(arduino, sprintf("start,%g,%g", dist, speed));
        end

        % 타이머 시작
        start(plotTimer);
        t = timer('StartDelay', targetDuration, ...
            'TimerFcn', @reportTime);
        start(t);

        % 센서 수신 루프
        sensorLoop(speed, dist);
    end

    % ------- 센서 루프 함수 -------
    function sensorLoop(speed, maxDist)
        while running
            if cas.NumBytesAvailable > 0
                raw = readline(cas);
                disp(['[센서 RAW] ' raw]);
                force_kg = parseForce(raw);
                if ~isempty(force_kg)
                    elapsed = toc(startTime);
                    distNow = min(elapsed * speed, maxDist);
                    forceN = force_kg * 9.80665;

                    distanceData(end+1) = distNow;
                    forceData(end+1) = forceN;
                    writematrix([elapsed, distNow, forceN], csvFile, 'WriteMode', 'append');
                end
            end
            pause(0.01);
        end
    end

    % ------- 그래프 업데이트 -------
    function updatePlot()
        clearpoints(plt);
        addpoints(plt, distanceData, forceData);
        drawnow limitrate;
    end

    % ------- 시간 보고 -------
    function reportTime(~, ~)
        corr = txtCorr.Value;
        actual = toc(startTime);
        corrected = actual * corr;
        lblTime.Text = sprintf('실제 소요시간: %.2f s', corrected);
        disp(['⏱️ 실제 소요시간: ' num2str(actual)]);
    end

    % ------- 스탑 -------
    function stopCompression()
        running = false;
        if ~isempty(arduino)
            writeline(arduino, "STOP");
        end
        stop(plotTimer);
        stop(t);
    end

    % ------- Back(복귀) 버튼 함수 -------
    function backMove()
        move_dist = -25;    % mm, 반대방향
        move_speed = 2;    % mm/s (10mm/1s)
        if ~isempty(arduino)
            writeline(arduino, sprintf("start,%g,%g", move_dist, move_speed));
            disp('[모터] Back(복귀) 명령 전송');
        end
    end

    % ------- Force 파싱 -------
    function val = parseForce(line)
        try
            if contains(line, 'kg')
                parts = split(line, ',');
                kgStr = strrep(parts{end}, 'kg', '');
                val = str2double(strrep(kgStr, '+', ''));
            else
                val = [];
            end
        catch
            val = [];
        end
    end
end
